cmake_minimum_required(VERSION 3.18)
project(mokoa_kylin_vector_bridge LANGUAGES CXX)
include(GNUInstallDirs)
find_package(nlohmann_json REQUIRED)
find_path(KYLIN_VECTOR_INCLUDE_DIR Database.h PATH_SUFFIXES kysdk-vector-engine-client REQUIRED)
find_library(KYLIN_VECTOR_LIBRARY NAMES kysdk-vector-engine-client REQUIRED)
add_library(mokoa_kylin_vector_bridge SHARED bridge.cpp)
target_compile_features(mokoa_kylin_vector_bridge PRIVATE cxx_std_17)
target_include_directories(mokoa_kylin_vector_bridge PRIVATE ${KYLIN_VECTOR_INCLUDE_DIR})
target_link_libraries(mokoa_kylin_vector_bridge PRIVATE ${KYLIN_VECTOR_LIBRARY} nlohmann_json::nlohmann_json)
set_target_properties(mokoa_kylin_vector_bridge PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
install(TARGETS mokoa_kylin_vector_bridge LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
